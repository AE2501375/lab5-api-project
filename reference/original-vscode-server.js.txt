const express = require('express');

const app = express();
app.use(express.json());

const PORT = 3000;
const HOSTNAME = '127.0.0.1';

// Trip data
const trips = [
    { id: 1, destination: 'Paris', country: 'France', cost: 400 },
    { id: 2, destination: 'Tokyo', country: 'Japan', cost: 800 },
    { id: 3, destination: 'Cairo', country: 'Egypt', cost: 150 },
];

// Get all trips
app.get('/trips', (req, res) => {
    res.status(200).json({
        status: 'success',
        message: 'Trips retrieved successfully',
        results: trips.length,
        data: trips
    });
});

// Create a new trip
app.post('/trips', (req, res) => {
    const { destination, country, cost, description } = req.body;
    // const destination = req.body.destination;
    // const country = req.body.country;
    // const cost = req.body.cost;
    // const description = req.body.description

    if (!destination || !country || !cost == null) {
        return res.status(400).json({
            status: 'fail',
            message: 'Please provide all required fields.'
        });
    }

    const newTrip = {
        id: trips.length + 1,
        destination,
        country,
        cost,
        description: description || 'No description available'
    };

    trips.push(newTrip);

    res.status(201).json({
        status: 'sucesss',
        message: 'Trip created successfully',
        data: newTrip
    });
});

app.get('/trips/id', (req, res) => {
    const id = Number(req.parms.id);

    for (let i = 0; i < trips.length; i++) {
        if (trips[i].id === id) {
            return res.status(200).json({
                status: 'success',
                data: trips[i]
            });
        }
    }

    return res.status(404).json({
        status: 'fail',
        message: 'Trip not found'
    });
});

app.delete('/trips/id', (req, res) => {
    const id = Number(req.parms.id);

    for (let i = 0; i < trips.length; i++) {
        if (trips[i].id === id) {
            const trip = trips.splice(i, 1);
            return res.status(200).json({
                status: 'success',
                data: trip
            });
        }
    }

    return res.status(404).json({
        status: 'fail',
        message: 'Trip not found'
    });
});
