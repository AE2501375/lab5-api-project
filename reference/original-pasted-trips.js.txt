const express = require("express");

const app = express();
app.use(express.json());

const trips = [
  { id: 1, destination: "Cairo",cost:300 },
  { id: 2, destination: "Paris",cost:301 },
  { id: 3, destination: "Rome" ,cost:302}
];

// GET all trips
app.get("/trips", (req, res) => {
  res.json(trips);
});

// GET a trip by ID
app.get("/trips/:id", (req, res) => {
  const id = Number(req.params.id);

  for (let i = 0; i < trips.length; i++) {
    if (trips[i].id === id) {
      return res.json(trips[i]);
    }
  }

  res.status(404).json({ message: "Trip not found" });
});

// POST a new trip
app.post("/trips", (req, res) => {
  const { destination,cost } = req.body;

  if (
    typeof destination !== "string" ||
    destination.trim() === ""
  ) {
    return res.status(400).json({
      message: "Provide a destination"
    });
  }

  const newTrip = {
    id: trips.length+1,
    destination: destination,
    cost:cost
  };

  trips.push(newTrip);

  res.status(201).json(newTrip);
});

// PUT: update an existing trip
app.put("/trips/:id", (req, res) => {
  const id = Number(req.params.id);

  for (let i = 0; i < trips.length; i++) {
    if (trips[i].id === id) {
      const { destination } = req.body;

      if (
        typeof destination !== "string" ||
        destination.trim() === ""
      ) {
        return res.status(400).json({
          message: "Provide a destination"
        });
      }

      trips[i].destination = destination.trim();

      return res.json(trips[i]);
    }
  }

  res.status(404).json({ message: "Trip not found" });
});

// DELETE a trip
app.delete("/trips/:id", (req, res) => {
  const id = Number(req.params.id);

  for (let i = 0; i < trips.length; i++) {
    if (trips[i].id === id) {
      trips.splice(i, 1);
      return res.status(204).send();
    }
  }

  res.status(404).json({ message: "Trip not found" });
});

app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});
